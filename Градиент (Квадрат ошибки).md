python
import time
def draw_track(x):
    width = 30
    pos = int((x + 10) / 20 * width)
    pos = max(0, min(width, pos))
    track = ["-"] * (width + 1)
    track[width // 2] = "|"
    track[pos] = "o"
    return "".join(track)
print("=" * 60)
print("ПРАКТИКУМ: ИССЛЕДОВАНИЕ ГРАДИЕНТНОГО СПУСКА")
print("=" * 60)
# -------------------------------------------------------------
# УРОВЕНЬ 1: СНАЙПЕР
# -------------------------------------------------------------
print("\n[УРОВЕНЬ 1: ТОЧНЫЙ СПУСК]")
print("Начальная точка: x = 8.0.")
print("Задача: подобрать learning_rate так, чтобы |x| < 0.1 строго на 3-м шаге.")
while True:
    try:
        lr = float(input("\nВведите learning_rate: "))
    except ValueError:
        print("Ошибка ввода. Введите число.")
        continue
    x = 8.0
    history = [x]
    
    for step in range(1, 4):
        grad = 2 * x
        x = x - lr * grad
        history.append(x)
        print(f"Шаг {step}: x = {x:6.2f} | {draw_track(x)}")
        time.sleep(0.15)
        
    if abs(history[1]) > 0.1 and abs(history[2]) > 0.1 and abs(history[3]) <= 0.1:
        print("Уровень 1 пройден: условие выполнено.")
        break
    else:
        print("Цель не достигнута на 3-м шаге. Повторите попытку.")
# -------------------------------------------------------------
# УРОВЕНЬ 2: ЗАТУХАЮЩИЕ КОЛЕБАНИЯ
# -------------------------------------------------------------
print("\n" + "=" * 60)
print("[УРОВЕНЬ 2: КОЛЕБАТЕЛЬНЫЙ РЕЖИМ]")
print("Начальная точка: x = 6.0.")
print("Задача: подобрать learning_rate, при котором алгоритм меняет знак позиции")
print("не менее 2 раз за 4 шага, но при этом сходится к |x| < 1.0.")
while True:
    try:
        lr = float(input("\nВведите learning_rate: "))
    except ValueError:
        print("Ошибка ввода. Введите число.")
        continue
    x = 6.0
    signs_changed = 0
    prev_x = x
    
    for step in range(1, 5):
        grad = 2 * x
        x = x - lr * grad
        if x * prev_x < 0:
            signs_changed += 1
        prev_x = x
        print(f"Шаг {step}: x = {x:6.2f} | {draw_track(x)}")
        time.sleep(0.15)
        
    if signs_changed >= 2 and abs(x) < 1.0:
        print("Уровень 2 пройден: режим затухающих колебаний зафиксирован.")
        break
    else:
        print("Условие не выполнено. Проверьте диапазон от 0.5 до 1.0.")
# -------------------------------------------------------------
# УРОВЕНЬ 3: ГРАНИЦА УСТОЙЧИВОСТИ

print("[УРОВЕНЬ 3: КРИТИЧЕСКАЯ ГРАНИЦА]")
print("Начальная точка: x = 5.0.")
print("Задача: найти точное значение learning_rate, при котором система")
print("переходит в режим незатухающих гармонических колебаний между +5.0 и -5.0.")
while True:
    try:
        lr = float(input("\nВведите learning_rate: "))
    except ValueError:
        print("Ошибка ввода. Введите число.")
        continue
    x = 5.0
    points = []
    
    for step in range(1, 4):
        grad = 2 * x
        x = x - lr * grad
        points.append(round(x, 2))
        print(f"Шаг {step}: x = {x:6.2f} | {draw_track(x)}")
        time.sleep(0.15)
        
    if points[0] == -5.0 and points[1] == 5.0:
        print("Уровень 3 пройден: найдена критическая точка устойчивости.")
        break
    else:
        print("Система либо затухает, либо расходится. Рассчитайте аналитически.")

print("Функция потерь изменена: Loss(x) = 4 * x^2. Градиент: grad = 8 * x.")
print("Начальная точка: x = 8.0.")
print("Задача: найти значение learning_rate для перехода в минимум за ровно 1 шаг.")
while True:
    try:
        lr = float(input("\nВведите learning_rate: "))
    except ValueError:
        print("Ошибка ввода. Введите число.")
        continue
    x = 8.0
    grad = 8 * x
    x = x - lr * grad
    print(f"Шаг 1: x = {x:6.2f} | {draw_track(x)}")
    
    if abs(x) < 0.001:
        print("Уровень 4 пройден: шаг скорректирован с учетом второй производной.")
        break
    else:
        print("Минимум не достигнут за 1 шаг. Проанализируйте выражение 1 - 8 * lr.")
# -------------------------------------------------------------
# УРОВЕНЬ 5: ОБРАТНАЯ ЗАДАЧА

print("\n" + "=" * 60)

print("Параметр шага фиксирован: learning_rate = 0.25.")
print("Задача: рассчитать начальную позицию x_start такую, чтобы")
print("ровно через 2 шага оптимизации координата x стала равна 2.0.")
while True:
    try:
        start_x = float(input("\nВведите начальную координату x: "))
    except ValueError:
        print("Ошибка ввода. Введите число.")
        continue
    x = start_x
    lr = 0.25
    
    for step in range(1, 3):
        grad = 2 * x
        x = x - lr * grad
        print(f"Шаг {step}: x = {x:6.2f} | {draw_track(x)}")
        time.sleep(0.15)
        
    if abs(x - 2.0) < 0.01:
